class Graph():
def unweighted(s):
q = Queue(NUM_VERTICES)
define v, w as vertex
q.enqueue( s )
s.dist = 0
while (!q.isEmpty()):
v = q.items[0]
q.dequeue()
v.known = true
for each w adjacent to v:
if w.dist == INFINITY :
w.dist = v.dist + 1
w.path = v
q.enqueue (w)
